	\subsection{Flow with lower bounds}
		Edge $u\to v$ with $l \le f \le c$: add edge with capacity $c-l$, $D[v] \mathrel{+}= l$, $D[u] \mathrel{-}= l$.
		New source $S'$, sink $T'$: for $D[x]>0$ add $S'\to x$ ($D[x]$), for $D[x]<0$ add $x \to T'$ ($-D[x]$).
		For an $s$-$t$ flow add $t\to s$ with capacity $\infty$. Feasible iff maxflow$(S',T') = \sum_{D>0} D$.
		Then max flow = flow on $t\to s$ + maxflow$(s,t)$ in the residual graph (after removing $t\to s$);
		for min flow, compute maxflow$(t,s)$ instead and subtract.

	\subsection{Theorems}
		K\H{o}nig: in bipartite graphs max matching = min vertex cover; max independent set = $V$ $-$ max matching;
		min edge cover = $V$ $-$ max matching (no isolated vertices).
		Hall: perfect matching of $L$ iff $|N(S)| \ge |S|$ for all $S \subseteq L$.
		Dilworth: in a DAG/poset, min chain cover (vertex-disjoint paths in the transitive closure) = max antichain
		= $V$ $-$ max matching in the split graph ($u_{out} \to v_{in}$ for each $u < v$).
		Min vertex-disjoint path cover of a DAG = $V$ $-$ max matching on its edges.

\section{Math}
	\subsection{Number of Spanning Trees}
		% I.e. matrix-tree theorem.
		% Source: https://en.wikipedia.org/wiki/Kirchhoff%27s_theorem
		% Test: stress-tests/graph/matrix-tree.cpp
		Create an $N\times N$ matrix \texttt{mat}, and for each edge $a \rightarrow b \in G$, do
		\texttt{mat[a][b]--, mat[b][b]++} (and \texttt{mat[b][a]--, mat[a][a]++} if $G$ is undirected).
		Remove the $i$th row and column and take the determinant; this yields the number of directed spanning trees rooted at $i$
		(if $G$ is undirected, remove any row/column).

	\subsection{Erd\H{o}s--Gallai theorem}
		% Source: https://en.wikipedia.org/wiki/Erd%C5%91s%E2%80%93Gallai_theorem
		% Test: stress-tests/graph/erdos-gallai.cpp
		A simple graph with node degrees $d_1 \ge \dots \ge d_n$ exists iff $d_1 + \dots + d_n$ is even and for every $k = 1\dots n$,
		\[ \sum _{i=1}^{k}d_{i}\leq k(k-1)+\sum _{i=k+1}^{n}\min(d_{i},k). \]
